# work_git
git init — ініціалізація локального репозиторію

git remote add origin git init — ініціалізація локального репозиторію

git remote add origin https://github.com/Oreshe/work_git.git — зв'язування з віддаленим репозиторієм

git pull origin main --allow-unrelated-histories — завантаження змін із віддаленого репозиторію та об'єднання розбіжних історій

git add . — індексування файлів для коміту

git commit -m "..." — створення коміту із збереженням стану

git push -u origin main — відправка локальних комітів у віддалений
